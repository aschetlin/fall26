\begin{problem}{1}
	Show that
	\begin{equation}
		\label{eq:p1-covariant-commutator}
		[\nabla_\mu,\nabla_\nu]V^\alpha
		= R^\alpha{}_{\lambda\mu\nu}V^\lambda - T^\lambda{}_{\mu\nu}\nabla_\lambda V^\alpha,
	\end{equation}
	where the Riemann curvature and torsion are the following tensors defined in terms of the connection
	\begin{equation}
		\label{eq:p1-torsion}
		T^\lambda{}_{\mu\nu} = \Gamma^\lambda{}_{\mu\nu}-\Gamma^\lambda{}_{\nu\mu},
	\end{equation}
	\begin{equation}
		\label{eq:p1-riemann}
		R^\alpha{}_{\beta\mu\nu}
		= \partial_\mu\Gamma^\alpha{}_{\nu\beta}-\partial_\nu\Gamma^\alpha{}_{\mu\beta}
		+\Gamma^\alpha{}_{\mu\lambda}\Gamma^\lambda{}_{\nu\beta}
		-\Gamma^\alpha{}_{\nu\lambda}\Gamma^\lambda{}_{\mu\beta}.
	\end{equation}
	(Recall the expression for the covariant derivative of a general tensor
	\begin{equation}
		\label{eq:p1-general-covariant-derivative}
		\begin{aligned}
			\nabla_\alpha T^{\mu\nu\cdots}{}_{\beta\gamma\cdots}
			= {} & \partial_\alpha T^{\mu\nu\cdots}{}_{\beta\gamma\cdots}
			+ \Gamma^\mu{}_{\alpha\lambda}T^{\lambda\nu\cdots}{}_{\beta\gamma\cdots}
			+ \Gamma^\nu{}_{\alpha\lambda}T^{\mu\lambda\cdots}{}_{\beta\gamma\cdots}+\cdots \\
			     & -\Gamma^\lambda{}_{\alpha\beta}T^{\mu\nu\cdots}{}_{\lambda\gamma\cdots}
			-\Gamma^\lambda{}_{\alpha\gamma}T^{\mu\nu\cdots}{}_{\beta\lambda\cdots}-\cdots).
		\end{aligned}
	\end{equation}
\end{problem}
\begin{proof}
	The commutator is defined as
	\begin{equation*}
		\begin{aligned}
      \comm{\nabla_{\mu}, \nabla_{\nu}}V^{\alpha} &= \nabla_{\mu}\nabla_{\nu}V^{\alpha} - \nabla_{\nu}\nabla_{\mu}V^{\alpha} \\
                               &= \partial_{\mu}\nabla_{\nu}V^{\alpha} + \Gamma_{\mu\lambda}^{\alpha}\nabla_{\nu}V^{\lambda} - \Gamma_{\mu\nu}^{\lambda}\nabla_{\lambda}V^{\alpha}
                                 - \partial_{\nu}\nabla_{\mu}V^{\alpha} - \Gamma_{\nu\lambda}^{\alpha}\nabla_{\mu}V^{\lambda} + \Gamma_{\nu\mu}^{\lambda}\nabla_{\lambda}V^{\alpha} \\[1ex]
                               &= \partial_{\mu}\left( \partial_{\nu}V^{\alpha} + \Gamma_{\nu\lambda}^{\alpha}V^{\lambda} \right) + \Gamma_{\mu\lambda}^{\alpha}\left( \partial_{\nu}V^{\lambda} + \Gamma_{\nu\sigma}^{\lambda}V^{\sigma} \right) - \Gamma_{\mu\nu}^{\lambda}\left( \partial_{\lambda}V^{\alpha} + \Gamma_{\lambda\sigma}^{\alpha}V^{\sigma} \right) \\
                               &\quad - \partial_{\nu}\left( \partial_{\mu}V^{\alpha} + \Gamma_{\mu\lambda}^{\alpha}V^{\lambda} \right) - \Gamma_{\nu\lambda}^{\alpha}\left( \partial_{\mu}V^{\lambda} + \Gamma_{\mu\sigma}^{\lambda}V^{\sigma} \right) + \Gamma_{\nu\mu}^{\lambda}\left( \partial_{\lambda}V^{\alpha} + \Gamma_{\lambda\sigma}^{\alpha}V^{\sigma} \right) \\[1ex]
                               &= \Gamma_{\nu\lambda}^{\alpha}\partial_{\mu}V^{\lambda} + V^{\lambda}\partial_{\mu}\Gamma_{\nu\lambda}^{\alpha} + \Gamma_{\mu\lambda}^{\alpha}\partial_{\nu}V^{\lambda} + \Gamma_{\mu\lambda}^{\alpha}\Gamma_{\nu\sigma}^{\lambda}V^{\sigma} - \Gamma_{\mu\nu}^{\lambda}\partial_{\lambda}V^{\alpha} - \Gamma_{\mu\nu}^{\lambda}\Gamma_{\lambda\sigma}^{\alpha}V^{\sigma} \\
                               &\quad -\Gamma_{\mu\lambda}^{\alpha}\partial_{\nu}V^{\lambda} - V^{\lambda}\partial_{\nu}\Gamma_{\mu\lambda}^{\alpha} - \Gamma_{\nu\lambda}^{\alpha}\partial_{\mu}V^{\lambda} - \Gamma_{\nu\lambda}^{\alpha}\Gamma_{\mu\sigma}^{\lambda}V^{\sigma} + \Gamma_{\nu\mu}^{\lambda}\partial_{\lambda}V^{\alpha} + \Gamma_{\nu\mu}^{\lambda}\Gamma_{\lambda\sigma}^{\alpha}V^{\sigma} \\[1ex]
                               &= V^{\lambda}\left( \partial_{\mu}\Gamma_{\nu\lambda}^{\alpha} - \partial_{\nu}\Gamma_{\mu\lambda}^{\alpha} \right) + V^{\sigma} \left( \Gamma_{\mu\lambda}^{\alpha}\Gamma_{\nu\sigma}^{\lambda} - \Gamma_{\nu\lambda}^{\alpha}\Gamma_{\mu\sigma}^{\lambda} - \Gamma_{\mu\nu}^{\lambda}\Gamma_{\lambda\sigma}^{\alpha} + \Gamma_{\nu\mu}^{\lambda}\Gamma_{\lambda\sigma}^{\alpha} \right) \\
                               &\quad + \Gamma_{\nu\lambda}^{\alpha}\left( \partial_{\mu}V^{\lambda} - \partial_{\mu}V^{\lambda} \right) + \Gamma_{\mu\lambda}^{\alpha}\left( \partial_{\nu}V^{\lambda} - \partial_{\nu}V^{\lambda} \right) + \partial_{\lambda}V^{\alpha}\left( \Gamma_{\nu\mu}^{\lambda} - \Gamma_{\mu\nu}^{\lambda} \right) \\[1ex]
                               &= V^{\lambda}\left( \partial_{\mu}\Gamma_{\nu\lambda}^{\alpha} - \partial_{\nu}\Gamma_{\mu\lambda}^{\alpha} + \Gamma_{\mu\sigma}^{\alpha}\Gamma_{\nu\lambda}^{\sigma} - \Gamma_{\nu\sigma}^{\alpha}\Gamma_{\mu\lambda}^{\sigma} \right) \\
                               &\quad - \Gamma_{\sigma\lambda}^{\alpha}V^{\lambda}\left( \Gamma_{\mu\nu}^{\sigma} - \Gamma_{\nu\mu}^{\sigma} \right) + \partial_{\lambda}V^{\alpha}\left( \Gamma_{\nu\mu}^{\lambda} - \Gamma_{\mu\nu}^{\lambda} \right)\\[1ex]
                               &= \tensor{R}{^{\alpha}_{\lambda\mu\nu}} V^{\lambda} - (\Gamma_{\mu\nu}^{\sigma} - \Gamma_{\nu\mu}^{\sigma})(\Gamma_{\sigma\lambda}^{\alpha} V^{\lambda} + \partial_{\sigma} V^{\alpha}) \\
      &= \tensor{R}{^{\alpha}_{\lambda\mu\nu}} V^{\lambda} - 2\Gamma_{[\mu\nu]}^{\sigma}(\nabla_{\sigma}V^{\alpha}) = \tensor{R}{^{\alpha}_{\lambda\mu\nu} }V^{\lambda} - \tensor{T}{^{\sigma}_{\mu\nu} }\nabla_{\sigma}V^{\alpha}.
		\end{aligned}
	\end{equation*}
\end{proof}

\begin{problem}{2}
	From now on we stick to the Levi-Civita connection.
	Show that the differential Bianchi identity \(\nabla_{[\rho}R_{\sigma\lambda]\mu\nu}=0\) implies the contracted Bianchi identity
	\begin{equation}
		\label{eq:p2-contracted-bianchi}
		\nabla^\nu G_{\mu\nu}=0,
	\end{equation}
	where \(G_{\mu\nu}=R_{\mu\nu}-\frac12 Rg_{\mu\nu}\) is the Einstein tensor.
\end{problem}
\begin{proof}
	Writing out the differential Bianchi identity and appealing to the antisymmetry of the Riemann tensor in the first two indices,
	\begin{equation}
		\label{eq:p2-explicit-bianchi}
		\begin{aligned}
			\nabla_{[\rho} R_{\sigma\lambda]\mu\nu} & = \frac{1}{3!} \left( \nabla_{\rho}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda} R_{\rho\sigma\mu\nu} + \nabla_{\sigma} R_{\lambda\rho\mu\nu} - \nabla_{\rho} R_{\lambda\sigma\mu\nu} - \nabla_{\lambda} R_{\sigma\rho\mu\nu} - \nabla_{\sigma}R_{\rho\lambda\mu\nu} \right) \\
			                                        & = \frac{1}{6} \left( 2\nabla_{\rho} R_{\sigma\lambda\mu\nu} + 2\nabla_{\lambda}R_{\rho\sigma\mu\nu} + 2\nabla_{\sigma} R_{\lambda\rho\mu\nu} \right) = 0
		\end{aligned}
	\end{equation}
	showing that
	\begin{equation}
		\label{eq:p2-bianchi-zero}
		\nabla_{\rho}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda}R_{\rho\sigma\mu\nu} + \nabla_{\sigma}R_{\lambda\rho\mu\nu} = 0.
	\end{equation}

	Now, we calculate
	\begin{equation}
		\label{eq:p2-diff-bianchi}
		\begin{aligned}
			\nabla^{\nu}G_{\mu\nu} &= g^{\lambda\nu}\nabla_{\lambda}G_{\mu\nu} = g^{\lambda\nu}\nabla_{\lambda}\left( R_{\mu\nu} - \frac{1}{2}Rg_{\mu\nu} \right) = g^{\lambda\nu}\left( \nabla_{\lambda}R_{\mu\nu} - \frac{1}{2}g_{\mu\nu}\nabla_{\lambda}R \right) \\
      &= \nabla^{\nu}R_{\mu\nu} - \frac{1}{2}\delta^{\lambda}_{\mu}\nabla_{\lambda}R = \nabla^{\nu}R_{\mu\nu} - \frac{1}{2}\nabla_{\mu}R.
		\end{aligned}
	\end{equation}
	where we have used metric compatibility, \(\nabla_{\alpha}g_{\beta\nu} = 0\).

  We now contract \cref{eq:p2-bianchi-zero} to get
  \begin{equation}
  \label{eq:8}
  \begin{aligned}
    g^{\rho\mu}\left( \nabla_{\rho}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda}R_{\rho\sigma\mu\nu} + \nabla_{\sigma}R_{\lambda\rho\mu\nu} \right) &= \nabla^{\mu}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda}R^{\mu}{}_{\sigma\mu\nu} + \nabla_{\sigma}R_{\lambda}{}^{\mu}{}_{\mu\nu} \\
    &= \nabla^{\mu}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda}R_{\sigma\nu} - \nabla_{\sigma}R_{\lambda\nu} = 0,
  \end{aligned}
\end{equation}
and contract again, so that
\begin{equation}
\label{eq:15}
\begin{aligned}
  g^{\lambda\nu}\left( \nabla^{\mu}R_{\sigma\lambda\mu\nu} + \nabla_{\lambda}R_{\sigma\nu} - \nabla_{\sigma}R_{\lambda\nu} \right) &= \nabla^{\mu}R_{\sigma\mu} + \nabla^{\nu}R_{\sigma\nu} - \nabla_{\sigma}R \\
  &= 2\nabla^{\mu}R_{\sigma\mu} - \nabla_{\sigma}R = 0.
\end{aligned}
\end{equation}
Finally, the expression obtained in \cref{eq:p2-diff-bianchi} is precisely \( \frac{1}{2} \) of the equation obtained in \cref{eq:15}, which proves that
\begin{equation}
\label{eq:16}
\nabla^{\nu}G_{\mu\nu} = 0.
\end{equation}
\end{proof}

\begin{problem}{4}
	Recall the parallel transport equation which tells you how to parallel transport a vector \(V^\mu\) along a curve \(x^\mu(\lambda)\),
	\begin{equation}
		\label{eq:p4-parallel-transport}
		\frac{d}{d\lambda}V^\mu+\Gamma^\mu{}_{\nu\rho}\frac{dx^\nu}{d\lambda}V^\rho=0.
	\end{equation}
	Consider a curve in Euclidean \(\mathbb{R}^2\) given in cartesian coordinates by \(x^1(\lambda)=\cos\lambda\), \(x^2(\lambda)=\sin\lambda\), \(\lambda\in[0,\frac\pi2]\).
	This curve traces out a quarter of the unit circle.
	Consider a vector which starts out as a unit vector pointing in the \(x^2\) direction: \(\left.V^\mu\right|_{\lambda=0}=(0,1)\).
	\begin{enumerate}[label=(\alph*)]
		\item Parallel transport on flat space conforms to our intuition that a vector remains constant as it is transported along the curve, so we expect \(V^\mu(\lambda)=(0,1)\) for all \(\lambda\) along the curve.
		      Verify that this indeed satisfies the parallel transport equation \cref{eq:p4-parallel-transport}.
          \begin{proof}
            The Christoffel symbols vanish in cartesian coordinates, so that \cref{eq:p4-parallel-transport} becomes
            \begin{equation}
            \label{eq:17}
            \dv{\lambda}V^{\mu} = 0.
          \end{equation}
          Since \( \dv{\lambda}V^{\mu} = (0,0) \), it is immediate that this satisfies the equation.
          \end{proof}
		\item Consider now the same setup in polar coordinates.
		      The equation for the curve is now
		      \begin{equation}
			      \label{eq:p4-polar-curve}
			      r(\lambda)=1,\qquad\theta(\lambda)=\lambda,
		      \end{equation}
		      and the initial vector is pointed in the \(\theta\) direction, \(\left.V^\mu\right|_{\lambda=0}=(0,1)\).
		      Solve the parallel transport equation \cref{eq:p4-parallel-transport} directly in polar coordinates with these initial conditions, using the connection coefficients you found in problem 3.
		      Show that the resulting vector points in the \(r\) direction at the end of the curve, \(\left.V^\mu\right|_{\lambda=\pi/2}=(1,0)\), and is in fact the same vector you get by parallel transporting in cartesian coordinates.
          \begin{proof}
            We have two equations to consider,
            \begin{equation}
            \label{eq:18}
            \begin{aligned}
              0 &= \dv{\lambda}V^r + \Gamma_{\nu\rho}^r \dv{x^{\nu}}{\lambda}V^{\rho} \\
                &= \dv{\lambda}V^r + \Gamma_{rr}^r\dv{x^r}{\lambda}V^r + \Gamma_{r\theta}^r\dv{x^r}{\lambda}V^{\theta} + \Gamma_{\theta r}^r\dv{x^{\theta}}{\lambda}V^r + \Gamma_{\theta\theta}^r\dv{x^{\theta}}{\lambda}V^{\theta} \\
                &= \dv{\lambda}V^r - r\dv{x^{\theta}}{\lambda}V^{\theta} \\
              &= \dv{\lambda}V^r - V^{\theta} = 0,
            \end{aligned}
          \end{equation}
          and
          \begin{equation}
          \label{eq:19}
          \begin{aligned}
            0 &= \dv{\lambda}V^{\theta} + \Gamma_{rr}^{\theta}\dv{x^r}{\lambda}V^r + \Gamma_{r\theta}^{\theta}\dv{x^r}{\lambda}V^{\theta} + \Gamma_{\theta r}^{\theta}\dv{x^{\theta}}{\lambda}V^r + \Gamma_{\theta\theta}^{\theta}\dv{x^{\theta}}{\lambda}V^{\theta} \\
            &= \dv{\lambda}V^{\theta} + V^r = 0.
          \end{aligned}
        \end{equation}

        This is a coupled system of 1st order ODEs, which has eigenvalues \( \pm i \) and corresponding eigenvectors \( v_1 = \begin{pmatrix}
1 \\
i
 \end{pmatrix}, v_2 = \begin{pmatrix}
1 \\
   -i
 \end{pmatrix} \).
 The general solution is \( \begin{pmatrix}
V^r \\
V^{\theta}
 \end{pmatrix} = c_1e^{i\lambda} \begin{pmatrix}
1 \\
i
 \end{pmatrix} + c_2e^{-i\lambda} \begin{pmatrix}
1 \\
-i
 \end{pmatrix}\).
 The initial condition means that \( c_1 + c_2 = 0 \) and \( i(c_1-c_2) = 1 \), giving \( c_1 = -i/2, c_2 = i/2 \).
 Therefore,
 \begin{equation}
 \label{eq:21}
 \begin{aligned}
   V^r &= \sin \lambda \\
   V^{\theta} &= \cos\lambda.
 \end{aligned}
\end{equation}

We calculate \( V^{\mu}(\lambda = \pi/2) = (1,0) \), as desired.
Transforming back to cartesian coordinates along the entire curve,
\begin{equation}
\label{eq:22}
\begin{aligned}
 V^{\prime 1}&=\frac{\partial x^1}{\partial r}V^r+\frac{\partial x^1}{\partial\theta}V^\theta
 =\cos\theta\,V^r-r\sin\theta\,V^\theta,\\
 V^{\prime 2}&=\frac{\partial x^2}{\partial r}V^r+\frac{\partial x^2}{\partial\theta}V^\theta
 =\sin\theta\,V^r+r\cos\theta\,V^\theta.
\end{aligned}
\end{equation}
Substituting \(r=1\), \(\theta=\lambda\), and \cref{eq:21} gives
\begin{equation}
\label{eq:23}
V^{\prime1}=\cos\lambda\sin\lambda-\sin\lambda\cos\lambda=0,\qquad
V^{\prime2}=\sin^2\lambda+\cos^2\lambda=1.
\end{equation}
Thus this is the same constant Cartesian vector as in part (a).
\end{proof}
	\end{enumerate}
\end{problem}

\begin{problem}{5}
	Consider the covariant laplacian of a scalar function \(\phi\),
	\begin{equation}
		\label{eq:p5-covariant-laplacian}
		\nabla^2\phi\equiv\nabla_\mu\nabla^\mu\phi=g^{\mu\nu}\nabla_\mu\nabla_\nu\phi.
	\end{equation}
	In cartesian coordinates on \(\mathbb{R}^2\), this reduces to the ordinary Laplacian \(\frac{\partial^2\phi}{\partial(x^1)^2}+\frac{\partial^2\phi}{\partial(x^2)^2}\).
	Using the connection coefficients you computed in problem 3b from the metric \cref{eq:p3-polar-metric}, find the Laplacian in polar coordinates and compare with the well-known expression
	\begin{equation}
		\label{eq:p5-polar-laplacian}
		\frac1r\frac{d}{dr}\left(r\frac{d\phi}{dr}\right)+\frac1{r^2}\frac{d^2\phi}{d\theta^2}.
	\end{equation}
\end{problem}
\begin{answer}
  We have that \( \laplacian{\varphi} \coloneq \nabla_{\mu}\nabla^{\mu} \varphi = g^{\mu\nu}\nabla_{\mu}\nabla_{\nu}\varphi\), so that
  \begin{equation}
  \label{eq:2}
  \begin{aligned}
    \laplacian{\varphi} &= g^{\mu\nu}\nabla_{\mu}\partial_{\nu}\varphi = g^{\mu\nu} \left( \partial_{\mu}\partial_{\nu}\varphi - \Gamma^{\alpha}_{\mu\nu}\partial_{\alpha}\varphi \right) \\
                  &= g^{rr} \left[ \pdv[2]{\varphi}{r} - \left( \Gamma_{rr}^r\partial_r\varphi + \Gamma_{rr}^{\theta}\partial_{\theta}\varphi \right) \right] + g^{\theta\theta} \left[ \pdv[2]{\varphi}{\theta} - \left( \Gamma_{\theta\theta}^r\partial_r\varphi + \Gamma_{\theta\theta}^{\theta}\partial_{\theta}\varphi \right) \right] \\
                  &= \left[ \pdv[2]{\varphi}{r} \right] + \frac{1}{r^2} \left[ \pdv[2]{\varphi}{\theta} + r\pdv{\varphi}{r} \right].
  \end{aligned}
\end{equation}

Now, noting that by the chain rule, \( \pdv*{r}(r\pdv*{\varphi}{r}) = r\pdv*[2]{\varphi}{r} + \pdv*{\varphi}{r}\), we get
\begin{equation}
\label{eq:14}
\nabla^2\varphi = \frac{1}{r}\pdv{r}\left( r\pdv{\varphi}{r} \right) + \frac{1}{r^2}\pdv[2]{\varphi}{\theta}.
\end{equation}
\end{answer}

\begin{problem}{6. Derivation of the Schwarzschild solution}
	The Schwarzschild solution was the first non-trivial exact solution to the Einstein field equations, derived by Karl Schwarzschild while he was in the German army fighting on the Russian front during World War I.
	Here you will go through his derivation.

	The most general static time-independent metric can be written in the form
	\begin{equation}
		\label{eq:p6-static-metric}
		ds^2=-A(r)dt^2+B(r)dr^2+r^2d\Omega^2,
	\end{equation}
	where \(d\Omega^2=d\theta^2+\sin^2\theta\,d\phi^2\) is the metric on a 2-sphere.
	When \(B(r)=A(r)=1\), this is flat Minkowski space in spherical coordinates.
	We would like our solution to approach flat space at infinity, so we demand the boundary conditions
	\begin{equation}
		\label{eq:p6-boundary-conditions}
		A(r)\xrightarrow{r\to\infty}1,\qquad B(r)\xrightarrow{r\to\infty}1.
	\end{equation}
	We are interested in vacuum solutions, i.e. solutions that exists outside any matter, so we set \(T_{\mu\nu}=0\) and look for solutions to the vacuum Einstein equations
	\begin{equation}
		\label{eq:p6-vacuum-einstein}
		G_{\mu\nu}=0.
	\end{equation}
	\begin{enumerate}[label=(\alph*)]
		\item Show that the vacuum Einstein equations \cref{eq:p6-vacuum-einstein} are equivalent to the vanishing of the Ricci tensor
		      \begin{equation}
			      \label{eq:p6-vacuum-ricci}
			      R_{\mu\nu}=0.
		      \end{equation}
          \begin{proof}
            Since the Einstein tensor is \( G_{\mu\nu} = R_{\mu\nu} - \frac{1}{2}Rg_{\mu\nu} \), the vacuum equations \cref{eq:p6-vacuum-einstein} are
            \begin{equation}
            \label{eq:9}
              0 = R_{\mu\nu} - \frac{1}{2}Rg_{\mu\nu} \iff R_{\mu\nu} = \frac{1}{2}Rg_{\mu\nu}.
          \end{equation}
          We can then contract both sides of the equation,
          \begin{equation}
          \label{eq:10}
          \begin{aligned}
            g^{\mu\nu}R_{\mu\nu} = R = \frac{1}{2}g^{\mu\nu}Rg_{\mu\nu} = 2R,
          \end{aligned}
        \end{equation}
        \begin{equation}
        \label{eq:11}
       \Rightarrow R = 2R,
      \end{equation}
      showing that the Ricci scalar satisfies \( R = 0 \).
      Substituting this back into \cref{eq:9} gives
      \begin{equation}
      \label{eq:12}
      R_{\mu\nu} = 0.
      \end{equation}
      Conversely, if \(R_{\mu\nu}=0\), then \(R=0\) and hence \(G_{\mu\nu}=0\).
          \end{proof}
		\item Compute the non-vanishing components of the Ricci tensor.
		      You should find
		      \begin{equation}
			      \label{eq:p6-ricci-components}
			      \begin{aligned}
				      R_{tt}           & =\frac{A''}{2B}-\frac{A'B'}{4B^2}-\frac{A'^2}{4AB}+\frac{A'}{rB},  \\
				      R_{rr}           & =-\frac{A''}{2A}+\frac{A'B'}{4AB}+\frac{A'^2}{4A^2}+\frac{B'}{rB}, \\
				      R_{\theta\theta} & =-\frac{rA'}{2AB}+\frac{rB'}{2B^2}-\frac1B+1,                      \\
				      R_{\phi\phi}     & =\sin^2\theta\,R_{\theta\theta}.
			      \end{aligned}
		      \end{equation}
          \begin{answer}
            We first calculate the Christoffel symbols,
            \begin{equation}
            \label{eq:4}
            \Gamma_{\mu\nu}^{\lambda} = \frac{1}{2}g^{\lambda\sigma} \left( \partial_{\mu}g_{\nu\sigma} + \partial_{\nu}g_{\mu\sigma} - \partial_{\sigma}g_{\mu\nu} \right).
            \end{equation}
            Since the metric is diagonal, any \(\Gamma_{\mu\nu}^{\lambda}\) with all three indices distinct vanish; moreover, if none of the indices are either \( r \) or \(\theta\), the symbols vanish.
            There are nine surviving components (up to symmetry in the lower indices), which are \( \Gamma_{rr}^r, \Gamma_{tt}^r, \Gamma_{\theta\theta}^r, \Gamma_{\phi\phi}^r, \Gamma_{r\theta}^{\theta}, \Gamma_{rt}^{t}, \Gamma_{r\phi}^{\phi}, \Gamma_{\theta\phi}^{\phi}, \Gamma_{\phi\phi}^{\theta} \).
            They are as follows
            \begin{equation}
            \label{eq:p6-christoffels}
            \begin{aligned}
              \Gamma_{tr}^t &= \frac{A^{\prime}}{2A}, \\
              \Gamma_{tt}^r &= \frac{A^{\prime}}{2B}, \quad \Gamma_{rr}^r = \frac{B^{\prime}}{2B}, \quad \Gamma_{\theta\theta}^r = - \frac{r}{B}, \quad \Gamma_{\phi\phi}^r = -\frac{r\sin^2\theta}{B}, \\
              \Gamma_{r\theta}^{\theta} &= \frac{1}{r}, \quad \Gamma_{\phi\phi}^{\theta} = -\sin\theta\cos\theta, \\
              \Gamma_{r\phi}^{\phi} &= \frac{1}{r}, \quad \Gamma_{\theta\phi}^{\phi} = \cot\theta.
            \end{aligned}
          \end{equation}

          Then, the Riemann tensor is
          \begin{equation}
          \label{eq:5}
          \tensor{R}{^{\mu}_{\alpha\nu\beta} } = \partial_{\nu}\Gamma_{\beta\alpha}^{\mu} - \partial_{\beta}\Gamma_{\nu\alpha}^{\mu} + \Gamma_{\nu\lambda}^{\mu}\Gamma_{\beta\alpha}^{\lambda} - \Gamma_{\beta\lambda}^{\mu}\Gamma_{\nu\alpha}^{\lambda},
          \end{equation}
          and the Ricci tensor is the contraction over the 1st and 3rd indices,
          \begin{equation}
          \label{eq:p6-ricci}
          R_{\alpha\beta} = \partial_{\rho}\Gamma_{\beta\alpha}^{\rho} - \partial_{\beta}\Gamma_{\rho\alpha}^{\rho} + \Gamma_{\rho\lambda}^{\rho}\Gamma_{\beta\alpha}^{\lambda} - \Gamma_{\beta\lambda}^{\rho}\Gamma_{\rho\alpha}^{\lambda}.
        \end{equation}
        To organize the sums, write
        \begin{equation}
          \label{eq:p6-traces}
          S\coloneq \Gamma^\rho{}_{\rho r}=\frac{A'}{2A}+\frac{B'}{2B}+\frac2r,
          \qquad \Gamma^\rho{}_{\rho\theta}=\cot\theta.
        \end{equation}
        Then
        \begin{equation}
          \label{eq:7}
          \begin{aligned}
            R_{tt}
            &=\left(\frac{A'}{2B}\right)'+S\frac{A'}{2B}
              -2\frac{A'}{2A}\frac{A'}{2B}\\
            &=\frac{A''}{2B}-\frac{A'B'}{4B^2}
              -\frac{A'^2}{4AB}+\frac{A'}{rB}.
          \end{aligned}
        \end{equation}
        Similarly,
        \begin{equation}
          \label{eq:3}
          \begin{aligned}
            R_{rr}
            &=\left(\frac{B'}{2B}\right)'-S'
              +S\frac{B'}{2B}
              -\left[\left(\frac{A'}{2A}\right)^2
              +\left(\frac{B'}{2B}\right)^2+\frac2{r^2}\right]\\
            &=-\frac{A''}{2A}+\frac{A'B'}{4AB}
              +\frac{A'^2}{4A^2}+\frac{B'}{rB}.
          \end{aligned}
        \end{equation}
        The angular components are
        \begin{equation}
          \label{eq:6}
          \begin{aligned}
            R_{\theta\theta}
            &=\left(-\frac rB\right)'+\csc^2\theta
              -\frac rB S+\frac2B-\cot^2\theta\\
            &=-\frac{rA'}{2AB}+\frac{rB'}{2B^2}-\frac1B+1,\\
            R_{\phi\phi}
            &=\sin^2\theta\left[\left(-\frac rB\right)'
              -\frac rB S+\frac2B\right]
              +(\sin^2\theta-\cos^2\theta)-\cos^2\theta+2\cos^2\theta\\
            &=\sin^2\theta\,R_{\theta\theta}.
          \end{aligned}
        \end{equation}
        Where we have used \(\csc^2\theta-\cot^2\theta=1\).
        The off-diagonal components vanish by symmetry; time reflection \(t\mapsto-t\) forces \(R_{ti}=0\) for spatial indices \(i\).
        Spherical symmetry forces the mixed radial-angular components to vanish and the angular block to be proportional to the sphere metric, so \(R_{r\theta}=R_{r\phi}=R_{\theta\phi}=0\).
          \end{answer}
		\item Form the combination \(\frac BA R_{tt}+R_{rr}=0\) and show that it reduces to the equation
		      \begin{equation}
			      \label{eq:p6-log-derivative-relation}
			      \frac{A'}A+\frac{B'}B=0.
		      \end{equation}
		      Show that this implies \(B=c/A\) for some constant \(c\).
		      Based on the boundary conditions \cref{eq:p6-boundary-conditions}, argue that we must have \(c=1\) so that
		      \begin{equation}
			      \label{eq:p6-reciprocal-relation}
			      B=\frac1A.
		      \end{equation}

          \begin{proof}
            The indicated combination gives
            \begin{equation}
              \label{eq:p6-combination}
              0=\frac BA R_{tt}+R_{rr}
              =\frac1r\left(\frac{A'}A+\frac{B'}B\right).
            \end{equation}
            We note that since \( (AB)^{\prime} = AB^{\prime} + A^{\prime}B \), \cref{eq:p6-combination} can be written as \( (AB)^{\prime}/AB \).
            Hence \((AB)'=0\), so \(AB=c\) and \(B=c/A\). Taking \(r\to\infty\) gives \(c=1\), and therefore \(B=1/A\).
          \end{proof}
		\item Plug \cref{eq:p6-reciprocal-relation} into \(R_{\theta\theta}\) and show that this gives the equation
		      \begin{equation}
			      \label{eq:p6-radial-equation}
			      -rA'-A+1=0.
		      \end{equation}
		      Solve this to obtain
		      \begin{equation}
			      \label{eq:p6-radial-solution}
			      A(r)=1-\frac{R_S}r,
		      \end{equation}
		      with \(R_S\) an integration constant.

          \begin{proof}
            Substituting \(B=1/A\) and \(B'=-A'/A^2\) yields
            \begin{equation}
              \label{eq:p6-angular-reduced}
              0=R_{\theta\theta}=-\frac{rA'}2-\frac{rA'}2-A+1
              =-rA'-A+1.
            \end{equation}
            Thus \((rA)'=1\), and integration gives \(rA=r+C\). Writing \(C=-R_S\), we obtain \(A=1-R_S/r\).
          \end{proof}
		\item Show that \cref{eq:p6-radial-solution} and \cref{eq:p6-reciprocal-relation} together solve \(R_{tt}=0\) and \(R_{rr}=0\) and thus give a solution to the full Einstein equations.

          \begin{proof}
            With \(B=1/A\), the remaining components simplify to
            \begin{equation}
              \label{eq:p6-remaining-ricci}
              R_{tt}=\frac A2\left(A''+\frac{2A'}r\right),\qquad
              R_{rr}=-\frac1{2A}\left(A''+\frac{2A'}r\right).
            \end{equation}
            For \(A=1-R_S/r\), we have \(A'=R_S/r^2\) and \(A''=-2R_S/r^3\), so the parentheses vanish. Together with \(R_{\theta\theta}=0\), \(R_{\phi\phi}=\sin^2\theta R_{\theta\theta}=0\), and the vanishing off-diagonal components, this proves \(R_{\mu\nu}=0\), hence \(G_{\mu\nu}=0\).
          \end{proof}
		\item We now have the metric
		      \begin{equation}
			      \label{eq:p6-schwarzschild-parameter-metric}
			      ds^2=-\left(1-\frac{R_S}r\right)dt^2+\frac{1}{1-\frac{R_S}r}dr^2+r^2d\Omega^2,
		      \end{equation}
		      which is a solution for any \(R_S\).
		      To fix the meaning of \(R_S\), expand the metric around flat space \(g_{\mu\nu}=\eta_{\mu\nu}+h_{\mu\nu}\).
		      For small \(h_{\mu\nu}\), we saw in class that we can identify the Newtonian potential as \(h_{00}=-2\Phi\).
		      Given that the Newtonian potential for a spherical mass is \(\Phi=-\frac{GM}r\), expand \cref{eq:p6-schwarzschild-parameter-metric} near \(r\to\infty\) where \(h_{\mu\nu}\) is small and the Newtonian approximation is good, and use this to identify
		      \begin{equation}
			      \label{eq:p6-schwarzschild-radius}
			      R_S=2GM.
		      \end{equation}
		      This is the Schwarzschild radius.

          \begin{answer}
            In spherical coordinates the flat background metric is
            \(\eta_{\mu\nu}=\operatorname{diag}(-1,1,r^2,r^2\sin^2\theta)\).
            For \(|R_S|/r\ll1\),
            \begin{equation}
              \label{eq:p6-weak-field}
              g_{tt}=-1+\frac{R_S}{r},\qquad
              g_{rr}=1+\frac{R_S}{r}+O\!\left(\frac{R_S^2}{r^2}\right).
            \end{equation}
            Consequently \(h_{00}=g_{tt}-\eta_{tt}=R_S/r\). Comparing with the Newtonian relation gives
            \begin{equation}
              \label{eq:p6-mass-identification}
              \frac{R_S}{r}=-2\Phi=\frac{2GM}{r},\qquad R_S=2GM.
            \end{equation}
          \end{answer}
	\end{enumerate}
	The final metric for the Schwarzschild solution thus reads
	\begin{equation}
		\label{eq:p6-schwarzschild-metric}
		ds^2=-\left(1-\frac{2GM}r\right)dt^2+\frac{1}{1-\frac{2GM}r}dr^2+r^2d\Omega^2.
	\end{equation}
\end{problem}
