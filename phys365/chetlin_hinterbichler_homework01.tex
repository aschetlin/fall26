\begin{problem}{5 (Carroll 1.7)}
  Suppose that in Minkowski space we have contravariant rank 2 tensor with components
  \begin{equation}
  \label{eq:1}
  \tensor{X}{^{\mu\nu}_{}} = \begin{pmatrix}
2   & 0 & 1 & -1 \\
-1  & 0 & 3 & 2 \\
-1  & 1 & 0 & 0 \\
-2  & 1 & 1 & -2
  \end{pmatrix},
  \end{equation}
  and a vector with components
  \begin{equation}
  \label{eq:2}
  \tensor{V}{^{\mu}_{}} = \begin{pmatrix}
-1  & 2 & 0 & -2
 \end{pmatrix}.
  \end{equation}
  Find the components of the following:
  \begin{enumerate}[label=(\alph*)]
    \item \( \tensor{X}{^{\mu}_{\nu}} \)
    \item \( \tensor{X}{_{\mu}^{\nu}} \)
    \item \( \tensor{X}{^{(\mu\nu)}_{}} \)
    \item \( \tensor{X}{^{}_{[\mu\nu]}} \)
    \item \( \tensor{X}{^{\lambda}_{\lambda}} \)
    \item \( \tensor{V}{^{\mu}_{}}\tensor{V}{^{}_{\mu}} \)
    \item \( \tensor{V}{^{}_{\mu}} \tensor{X}{^{\mu\nu}_{}}\)
  \end{enumerate}
\end{problem}
