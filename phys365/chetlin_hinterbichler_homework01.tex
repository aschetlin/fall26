\begin{problem}{1 (Precession of perihelion of Mercury)}

\begin{problem}{2 (Bending of light in Newtonian gravity)}
  Consider a particle with inertial and gravitational masses \( m_I, m_G \) moving in the gravitational field of the Sun.
  The equation of motion is
  \begin{equation}
  \label{eq:9}
  m_I \vec{\ddot{x}} = - m_G \nabla \phi(\vec{x}), \quad \phi(\vec{x}) = - \frac{GM}{| \vec{x} |},
  \end{equation}
  where \( M \) is the mass of the sun.

  The particles path in the absence of \(\phi\) is \( \vec{x}_{(0)}(t) \) which obeys
  \begin{equation}
  \label{eq:10}
  \vec{\ddot{x}}_{(0)} = 0.
\end{equation}

Let the unperturbed path be the one taken by a photon moving at \( c \) parallel to the \( z \) axis, at an impact parameter \( b \) as measured from the origin,
\begin{equation}
\label{eq:11}
\vec{x}_{(0)}(t) = (b,0, ct).
\end{equation}

\begin{enumerate}[label=(\alph*)]
  \item Find the equation of motion satisfied by the first order correction to the free motion by expanding
        \begin{equation}
        \label{eq:12}
        \vec{x} = \vec{x}_{(0)} + \vec{\delta x},
      \end{equation}
        where \( \vec{\delta x} \) is of the same order as \(\phi\) and substituting into \cref{eq:9}.

  \item We can integrate \cref{eq:12} to obtain
        \begin{equation}
        \label{eq:13}
        \vec{\delta \dot{x}} = - \frac{m_{G}}{m_{I}} \int\limits_{-\infty}^{\infty} dt \nabla \phi(x_{(0)}(t)).
        \end{equation}
        Referring to the definitions in figure 2, evaluate the deflection angle \(\vec{\alpha} = - \frac{\vec{\delta \dot{x}}}{\left| \vec{\dot{x}}_{(0)} \right|}\) for the Newtonian potential and show that its magnitude is
        \begin{equation}
        \label{eq:14}
        \left| \vec{\alpha} \right| = \frac{m_G}{m_I} \frac{2GM}{c^2 b}.
        \end{equation}
        \item Evaluate the approximate size of the deflection angle in arcseconds for stars observed near the solar disk during an eclipse.
\end{enumerate}
\end{problem}
