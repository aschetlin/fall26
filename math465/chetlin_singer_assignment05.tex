\begin{problem}{3.2.5}
  Show that the mean curvature \( H \) at \( p \in S \) is given by
  \begin{equation}
  \label{eq:1}
  H = \frac{1}{\pi} \int\limits_0^{\pi} k_n(\theta) d \theta,
\end{equation}
where \( k_n(\theta) \) is the normal curvature at \( p \) along a direction making an angle \(\theta\) with a fixed direction.
\end{problem}
\begin{proof}
  Let \( L: T_pS \rightarrow T_pS \) be the Weingarten map (equivalently, \( -dN_p \) for \( N \) the Gauss map), which, by the spectral theorem for finite dimensional operators, has an orthonormal set of eigenvectors \( e_1, e_2 \), the principal directions of curvature of \( S \) at \( p \), with associated principal curvatures \( k_1 \) and \( k_2 \).

  If \(\alpha: I \rightarrow S\) is any curve on the surface with \(\alpha(0)=p\) and \( \alpha^{\prime}(0) = V \), then \( V \in T_pS \) and we can write \( V = \cos\theta e_1 + \sin\theta e_2 \).
  The normal curvature along \( V \) is \( k_n = D_VV \cdot \nu = \langle L(V), V \rangle \).
  \begin{equation}
  \label{eq:9}
  \begin{aligned}
    \langle L(V), V \rangle &= \langle L(\cos\theta e_1 + \sin\theta e_2), \cos\theta e_1 + \sin\theta e_2 \rangle \\
                &= \langle k_1\cos\theta e_1 + k_2\sin\theta e_2, \cos\theta e_1 + \sin\theta e_2 \rangle \\
    &= k_1\cos^2\theta + k_2 \sin^2\theta.
  \end{aligned}
\end{equation}

Therefore, we can write \cref{eq:1} as
\begin{equation}
\label{eq:10}
\begin{aligned}
  \frac{1}{\pi} \int\limits_0^{\pi} (k_1\cos^2\theta + k_2\sin^2\theta) d\theta &= \frac{1}{\pi} \left[ \int\limits_0^{\pi} k_1 \frac{1 + \cos2\theta}{2}d\theta + \int\limits_0^{\pi} k_2 \frac{1-\cos2\theta}{2}d\theta \right] \\
  &= \frac{1}{\pi} \left( k_1( \frac{\pi}{2} ) + k_2( \frac{\pi}{2} ) \right) = \frac{k_{1} + k_2}{2}.
\end{aligned}
\end{equation}
\end{proof}

\begin{problem}{3.2.8}
  Describe the region of the unit sphere covered by the image of the Gauss map of the following surfaces:
  \begin{enumerate}[label=(\alph*)]
    \item Paraboloid of revolution \( z = x^2 + y^2 \),
          \begin{answer}
            The paraboloid can be parametrized as
            \begin{equation}
            \label{eq:12}
            \mathbf{x}(u,v) = (u, v, u^2 + v^2),
          \end{equation}
          for which the unit normal vector is
          \begin{equation}
          \label{eq:13}
          \nu = \frac{\mathbf{x}_u \wedge \mathbf{x}_v}{|\mathbf{x}_u \wedge \mathbf{x}_v |} = \frac{(-2u, -2v, 1)}{\sqrt{4 u^2 + 4 v^2 + 1}}.
        \end{equation}
        Clearly, \(\nu^1\) and \(\nu^2\) can range over all possible values on \( S^2 \), so that our only limitation is that of \( \nu^3 \); in particular, \(\nu^3 > 0\), so that the Gauss map covers the open upper hemisphere (or lower, depending on choice of orientation).
          \end{answer}
    \item Hyperboloid of revolution \( x^2 + y^2 - z^2 = 1 \),
          \begin{answer}
            Parametrize the upper half of the hyperboloid as \( \mathbf{x}(u,v) = (u, v, \sqrt{x^2 + y^2 - 1}) \),
            so that
            \begin{equation}
            \label{eq:14}
            N = \left(\frac{-u}{\sqrt{u^2 + v^2 -1}}, \frac{-v}{\sqrt{u^2 + v^2 - 1}}, 1\right),
          \end{equation}
          and the unit normal is then
          \begin{equation}
          \label{eq:15}
          \nu = \frac{(-u, -v, \sqrt{u^2 + v^2  - 1})}{\sqrt{2(u^2 + v^2) - 1}}.
        \end{equation}
        The \(\nu^3\) component satisfies
        \begin{equation}
        \label{eq:16}
        (\nu^3)^2 = \frac{u^2 + v^2 - 1}{2(u^2 + v^2) - 1},
      \end{equation}
      where \( u^2 + v^2 > 1 \), this ranges as \( 0 < (\nu^3)^2 < \frac{1}{2} \), and hence \( 0 < \nu^3 < \frac{1}{\sqrt{2}} \).
      The lower open hemisphere similarly ranges over \( - \frac{1}{\sqrt{2}} < \nu^3 < 0 \).
      Therefore, the image of the Gauss map is the open band on the \( S^3 \) with polar angle \( \frac{\pi}{4} < \theta < \frac{3\pi}{4} \).
          \end{answer}
    \item Catenoid \( x^2 + y^2 = \cosh^2 z \).
          \begin{answer}
            We may parametrize the Catenoid as in \cref{eq:3}, \( \mathbf{x}(u,v) = (\cosh v\cos u, \cosh v \sin u, v) \), whence the unit normal is
            \begin{equation}
            \label{eq:17}
            \begin{aligned}
              \nu &= \frac{1}{\cosh^2 v} \left( \cosh v \cos u, \cosh v \sin u, -\cosh v \sinh v \right) \\
              &= \left( \frac{\cos u}{\cosh u}, \frac{\sin u}{\cosh u}, -\tanh v \right).
            \end{aligned}
          \end{equation}
          Therefore the \(\nu^3\) component ranges over \( -1 < \nu^3 < 1 \), so that the image of the Gauss map is \( S^3 \setminus \{(0,0,1), (0,0,-1)\} \).
          \end{answer}
  \end{enumerate}
\end{problem}

\begin{problem}{3.2.13 (Theorem of Beltrami-Enneper)}
  Prove that the absolute value of the torsion \(\tau\) at a point of an asymptotic curve, whose curvature is nowhere zero, is given by
  \begin{equation}
  \label{eq:2}
  | \tau | = \sqrt{- K},
\end{equation}
where \( K \) is the Gaussian curvature of the surface at a given point.
\end{problem}
\begin{proof}
  Suppose \(\alpha \subset S \) is an (arc-length parametrized) asymptotic curve so that for all \( p \in S, \alpha(t) = p \), we have
  \begin{equation}
 \mathrm{I\!I}(\alpha^{\prime}(t)) = \langle L(T), T \rangle = \langle -D_T\nu, T \rangle = \langle D_T T, \nu \rangle = \langle T^{\prime}, \nu \rangle = \langle kN, \nu \rangle = 0,
\end{equation}
and since \( k \neq 0 \), \( \langle N, \nu \rangle = 0 \).
The \( TN \) plane is thus the tangent plane \( T_pS \), which means \(\nu\) is the binormal vector \( \nu = B \), so that \(D_{T}\nu = \tau N \).
We can calculate \( L_i^j \) in the \( \{T,N\} \) basis.
\( L_i^j = g^{jk}\mathrm{I\!I}_{ki} \).
We have that \( \langle L(T), T \rangle = 0, \langle L(T), N \rangle = \langle -D_T\nu, N \rangle = -\tau \), and \( \langle L(N), N \rangle = k_n(N) \).
Since our basis is orthonormal, raising indices doesn't impact the matrix, so that \( L_1^1 = 0, L_1^2 = L_2^1 = -\pm \tau, L_2^2 = k_n(N) \).
Then the Gauss curvature is \( L_1^1L_2^2 - (L_1^2)^2 = -\tau^2 = K \), and thus \( | \tau | = \sqrt{- K} \), as desired.
\end{proof}

\begin{problem}{3.3.3}
  Determine the asymptotic curves of the catenoid
  \begin{equation}
  \label{eq:3}
  \mathbf{x}(u,v) = (\cosh v \cos u, \cosh v \sin u, v).
  \end{equation}
\end{problem}
\begin{proof}
  First of all,
  \begin{equation}
  \label{eq:29}
  \begin{aligned}
    \mathbf{x}_u = (-\cosh v \sin u, \cosh v \cos u, 0), \quad \mathbf{x}_v = (\sinh v \cos u, \sinh v \sin u, 1),
  \end{aligned}
\end{equation}
and
\begin{equation}
\label{eq:30}
\begin{aligned}
  \mathbf{x}_{uu} &= (-\cosh v \cos u, -\cosh v \sin u, 0) \\
  \mathbf{x}_{uv} &= (-\sinh v \sin u, \sinh v \cos u, 0) \\
  \mathbf{x}_{vv} &= (\cosh v \cos u, \cosh v \sin u, 0),
\end{aligned}
\end{equation}
and the unit normal is
\begin{equation}
\label{eq:31}
\begin{aligned}
\nu &= \frac{\left( \cosh v \cos u, \cosh v \sin u, -\cosh v \sinh v \right)}{\cosh^2 v} \\
&= \frac{(\cos u, \sin u, -\sinh v)}{\cosh v}.
\end{aligned}
\end{equation}
We can now calculate the coefficients of the second fundamental form as
\begin{equation}
\label{eq:32}
\begin{aligned}
  e &= \langle \mathbf{x}_{uu}, \nu \rangle = -1(\cos^2 u + \sin^2 u) = -1 \\
  f &= \frac{1}{\cosh v}(-\sinh v \cos u \sin u + \sinh v \cos u \sin u) = 0 \\
  g &= 1(\cos^2 u + \sin^2 u) = 1.
\end{aligned}
\end{equation}
Now we solve for curves \(\alpha(t) = (u(t),v(t))\) for which \( e(u^{\prime})^2 + 2f(u^{\prime} v^{\prime}) + g(v^{\prime})^2 = 0 \), that is,
\begin{equation}
\label{eq:33}
\begin{aligned}
0 &= - (u^{\prime})^2 + (v^{\prime})^2,
\end{aligned}
\end{equation}
so that \( u^{\prime} = \pm v^{\prime} \), giving the asymptotic curves as \( u \pm v = \text{const}. \)
\end{proof}

\begin{problem}{3.3.5}
  Consider the parametrized surface (Enneper's surface)
  \begin{equation}
  \label{eq:4}
  \mathbf{x}(u,v) = (u - \frac{u^3}{3} + uv^2, v - \frac{v^3}{3} + vu^2, u^2 - v^2),
\end{equation}
and show that
\begin{enumerate}[label=(\alph*)]
  \item The coefficients of the first fundamental form are
        \begin{equation}
        \label{eq:5}
        E = G = (1 + u^2 + v^2)^2, \quad F = 0.
      \end{equation}
        \begin{proof}
          We first find our coordinate basis vectors,
          \begin{equation}
          \label{eq:8}
          \mathbf{x}_u = \left( 1 - u^2 + v^2, 2vu, 2u \right), \quad \mathbf{x}_v = \left( 2uv, 1 - v^2 + u^2, -2v \right).
        \end{equation}
        Then
        \begin{equation}
          \begin{aligned}
            E &= \langle \mathbf{x}_u, \mathbf{x}_u \rangle = (1-u^2 + v^2)^2 + 4v^2 u^2 + 4 u^2 \\
              &= 1 + u^4 + v^4 - 2u^2 + 2v^2 -2u^2 v^2 + 4 v^2 u^2 + 4 u^2 \\
            &= 1 + u^4 + v^4 + 2 v^2 + 2 u^2 + 2 v^2 u^2 = (1 + u^2 + v^2)^2,
          \end{aligned}
        \end{equation}
        and
        \begin{equation}
        \label{eq:11}
        \begin{aligned}
          G &= \langle \mathbf{x}_v, \mathbf{x}_v \rangle = 4 u^2 v^2 + (1 - v^2 + u^2)^2 + 4 v^2 \\
            &= 1 + v^4 + u^4 -2v^2 + 2 u^2 - 2 v^2 u^2 + 4 u^2 v^2 + 4 v^2 \\
          &= 1 + v^4 + u^4 + 2 v^2 + 2 u^2 + 2 u^2 v^2 = (1 + v^2 + u^2)^2.
        \end{aligned}
      \end{equation}
      Finally,
      \begin{equation}
      \label{eq:18}
      \begin{aligned}
        F &= \langle \mathbf{x}_u, \mathbf{x}_v \rangle = 2uv(1- u^2 + v^2) + 2uv( 1 - v^2 + u^2) - 4 u v \\
        &= 4 uv - 4 uv = 0.
      \end{aligned}
      \end{equation}
        \end{proof}
  \item The coefficients of the second fundamental form are
        \begin{equation}
        \label{eq:6}
        e = 2, \quad g = -2, \quad f = 0.
      \end{equation}
        \begin{proof}
          First of all, the second order partials are
          \begin{equation}
          \label{eq:19}
          \begin{aligned}
            \mathbf{x}_{uu} = \left( -2u, 2v, 2 \right), \ \mathbf{x}_{uv} = \mathbf{x}_{vu} = \left( 2v, 2u, 0 \right), \ \mathbf{x}_{vv} = \left( 2u, -2v, -2 \right),
          \end{aligned}
        \end{equation}
        and the surface normal is
        \begin{equation}
        \label{eq:20}
        \begin{aligned}
        \nu &= \frac{\mathbf{x}_u \wedge \mathbf{x}_v }{| \mathbf{x}_u \wedge \mathbf{x}_v |}.
        \end{aligned}
      \end{equation}
      We calculate
      \begin{equation}
        \begin{aligned}
          &\left(-4v^2u - 2u(1-v^2 + u^2), 2v(1 - usr + v^2) + 4 u^2v, (1- u^2 + v^2)(1 - v^2 + u^2) -4 u^2 v^2\right) \\
          &= \left(-2u(1 + v^2 + u^2), 2v(1 + v^2 + u^2), 1 - (u^2 + v^2)^2\right) \\
          &= (1 + v^2 + u^2)(-2u, 2v, 1 - u^2 - v^2),
        \end{aligned}
      \end{equation}
      and
      \begin{equation}
      \label{eq:21}
      \begin{aligned}
        | \mathbf{x}_u \wedge \mathbf{x}_v | &= (1 + u^2 + v^2) \sqrt{4 u^2 + 4 v^2 + (1 - u^2 - v^2)^2} \\
                                        &= (1 + u^2 + v^2) \sqrt{4 u^2 + 4 v^2 + 1 + u^{4} + v^4 - 2 u^2 - 2 v^2 + 2 u^2 v^2} \\
                                        &= (1 + u^2 + v^2) \sqrt{1 + 2(u^2 + v^2) + (u^2 + v^2)^2} \\
        &= (1 + u^2 + v^2) \sqrt{(1 + u^2 + v^2)^2} = (1 + u^2 + v^2)^2.
      \end{aligned}
    \end{equation}
    Therefore,
    \begin{equation}
    \label{eq:22}
    \begin{aligned}
      \nu = \frac{(-2u, 2v, 1 - u^2 - v^2)}{(1 + u^2 + v^2)}.
    \end{aligned}
  \end{equation}
  Then we can calculate the coefficients as
  \begin{equation}
  \label{eq:23}
  \begin{aligned}
    e &= \langle \mathbf{x}_{uu}, \nu \rangle = \left\langle (-2u, 2v, 2), \left(\frac{-2u}{1 + u^2 + v^2}, \frac{2v}{1 + u^2 + v^2}, \frac{1 - u^2 - v^2}{1 + u^2 + v^2}\right) \right\rangle \\
    &= \frac{4 u^2 + 4 v^2 + 2 - 2u^2 - 2 v^2}{1 + u^2 + v^2} = \frac{2(1 + u^2 + v^2)}{1 + u^2 + v^2} = 2,
  \end{aligned}
\end{equation}
\begin{equation}
\label{eq:25}
\begin{aligned}
  g &= \langle \mathbf{x}_{vv}, \nu \rangle = \left\langle (2u, -2v, -2), \left(\frac{-2u}{1 + u^2 + v^2}, \frac{2v}{1 + u^2 + v^2}, \frac{1 - u^2 - v^2}{1 + u^2 + v^2} \right) \right\rangle \\
  &= \frac{-4u^2 - 4 v^2 -2 + 2 u^2 + 2 v^2}{1 + u^2 + v^2} = \frac{-2(1 + u^2 + v^2)}{1 + u^2 + v^2} = -2,
\end{aligned}
\end{equation}
and
\begin{equation}
\label{eq:24}
\begin{aligned}
  f &= \langle \mathbf{x}_{uv}, \nu \rangle = \left\langle (2v, 2u, 0), \left(\frac{-2u}{1 + u^2 + v^2}, \frac{2v}{1 + u^2 + v^2}, \frac{1 - u^2 - v^2}{1 + u^2 + v^2} \right) \right\rangle \\
  &= \frac{-4uv + 4uv}{1 + u^2 + v^2} = 0.
\end{aligned}
\end{equation}
  \end{proof}
  \item The principal curvatures are
        \begin{equation}
        \label{eq:7}
        k_1 = \frac{2}{(1 + u^2 + v^2)^2}, \quad k_2 = - \frac{2}{(1 + u^2 + v^2)^2}.
      \end{equation}
        \begin{proof}
          The principal curvatures are the eigenvalues of the Weingarten map.
          We have
          \begin{equation}
          \label{eq:26}
          \mathrm{I\!I} \doteq \begin{pmatrix}
2  & 0 \\
0  & -2
          \end{pmatrix},
        \end{equation}
        and then \( L_i^j = g^{jk}\mathrm{I\!I}_{ik} \), which written in matrix form is
        \begin{equation}
        \label{eq:27}
        \begin{aligned}
        \begin{pmatrix}
L_1^1  & L_1^2 \\
L_2^1  & L_2^2
 \end{pmatrix} &= \begin{pmatrix}
g^{11}  & g^{12} \\
g^{21}  & g^{22}
 \end{pmatrix} \begin{pmatrix}
e  & f \\
f  & g
 \end{pmatrix} \\
          &= \begin{pmatrix}
(1 + u^2 + v^2)^{-2}  & 0 \\
0                     & (1 + u^2 + v^2)^{-2}
 \end{pmatrix} \begin{pmatrix}
2  & 0 \\
0  & -2
 \end{pmatrix} \\
          &= \begin{pmatrix}
2(1 + u^2 + v^2)^{-2}  & 0 \\
0                      & -2(1 + u^2 + v^2)^{-2}
 \end{pmatrix}.
        \end{aligned}
      \end{equation}
      This matrix is already diagonalized, so the eigenvalues lie on the diagonal, which match with \cref{eq:7}.
\end{proof}
  \item The lines of curvature are the coordinate curves.
        \begin{proof}
          Since the matrix is globally diagonalized, that implies the basis vectors are the eigenvectors, that is, the principal directions.
          Hence, the lines of curvature are the coordinate curves.
        \end{proof}
  \item The asymptotic curves are \( u + v = \text{const.} \), \( u - v = \text{const.} \)
        \begin{proof}
          We seek curves \(\alpha \subset S\) such that for all \( t \), \( \mathrm{I\!I}(\alpha^{\prime}(t)) = 0 \).
          Write \( \alpha^{\prime}(t) = \mathbf{x}_u V^u(t) + \mathbf{x}_v V^v(t) \) where \( V^{\mu} \) are the component functions of \( \alpha^{\prime}(t) \).
          Then we have that
          \begin{equation}
          \label{eq:28}
          \begin{aligned}
            \mathrm{I\!I}(\alpha^{\prime}(t)) &= e (V^u)^2 + 2f(V^uV^v) + g(V^v)^2 \\
            &= 2 (V^{u})^2 - 2(V^v)^2 = 0.
          \end{aligned}
        \end{equation}
        This is satisfied when \( V^{u} = \pm V^v \), or, in other words, \( V^u \pm V^v = 0 \).
        Note that \( V^u \pm V^v = \frac{\mathrm{d} }{\mathrm{d} t}(\alpha^1(t) \pm \alpha^2(t)) = 0\), so that \( \alpha^1 \pm \alpha^2 = \text{const.} \), as desired.
        \end{proof}
\end{enumerate}
\end{problem}
